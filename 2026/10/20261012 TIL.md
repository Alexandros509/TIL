상태(State): 컴포넌트 내부에서 시간에 따라 변하는 반응형 데이터

- Setter 함수: State 값을 변경하고 리액트에 화면 갱신(리렌더링)을 요청하는 전용 함수
- 일반 JS 변수와 달리 리액트는 상태의 참조 변경을 기준으로 렌더링 여부를 결정

~~~typescript
// [app/components/Counter.tsx]
'use client'; // 클라이언트 상호작용을 위해 선언합니다.

import { useState } from 'react';

export default function Counter() {
  // count 상태와 setCount 변경 함수 선언 (초기값: 0)
  const [count, setCount] = useState<number>(0);

  return (
    <div className="p-4 border rounded-lg max-w-xs text-center bg-white shadow-sm">
      <p className="text-xl font-bold mb-2 text-gray-800">현재 카운트: {count}</p>
      {/* 버튼 클릭 시 Setter 함수를 실행하여 1을 증가시킵니다. */}
      <button
        onClick={() => setCount(count + 1)}
        className="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600 transition-colors"
      >
        +1 증가
      </button>
    </div>
  );
}
~~~

**Automatic Batching**: 한 이벤트 핸들러 안에서 상태 변경이 여러 번 일어나도 리액트가 모아서 한 번에 처리(1회 리렌더링)

- 일반 업데이트 `setCount(count + 1)`: 현재 렌더링 시점의 고정된 State 값을 참조 → 연속 호출해도 1번만 적용
- **함수형 업데이트** `setCount(prev => prev + 1)`: 이전 상태의 최신값(prev)을 매개변수로 받아 순차적으로 안전하게 계산

~~~typescript
// [app/components/BatchTest.tsx]
'use client';

import { useState } from 'react';

export default function BatchTest() {
  const [number, setNumber] = useState<number>(0);

  // 일반 업데이트: 동일한 스냅샷 값(0)을 참조하므로 3번 실행해도 결과는 +1이 됩니다.
  const handleNormalAdd = () => {
    setNumber(number + 1);
    setNumber(number + 1);
    setNumber(number + 1);
  };

  // 함수형 업데이트: 최신 이전 값(prev)을 순차적으로 전달받아 정확히 +3이 됩니다.
  const handleFunctionalAdd = () => {
    setNumber((prev) => prev + 1);
    setNumber((prev) => prev + 1);
    setNumber((prev) => prev + 1);
  };

  return (
    <div className="p-4 border rounded-lg max-w-md space-y-3 bg-white shadow-sm">
      <p className="text-lg font-bold text-gray-800">결과 숫자: {number}</p>
      <div className="flex gap-2">
        <button onClick={handleNormalAdd} className="p-2 bg-orange-500 text-white rounded hover:bg-orange-600 transition-colors">
          일반 +3 테스트
        </button>
        <button onClick={handleFunctionalAdd} className="p-2 bg-green-500 text-white rounded hover:bg-green-600 transition-colors">
          함수형 +3 테스트
        </button>
        <button onClick={() => setNumber(0)} className="p-2 bg-gray-500 text-white rounded hover:bg-gray-600 transition-colors">
          초기화
        </button>
      </div>
    </div>
  );
}
~~~

- [일반 +3 테스트]: 1씩만 증가 / [함수형 +3 테스트]: 정확히 3씩 증가

**불변성**(Immutability): 원본을 직접 바꾸지 않고 새 복사본(새 주소)을 만들어 유지하는 원칙

- 리액트는 메모리 주소만 확인하는 **얕은 비교**(Shallow Compare)로 변경을 판단
- 원본 배열을 직접 수정(push, splice 등)하면 주소가 그대로라 리액트가 변경을 감지하지 못함
- 불변성 배열 조작
  - 추가: `[...prev, newItem]` (전개 연산자로 새 배열) / `prev.concat(newItem)` (새 배열 반환)
  - 수정: `prev.map(...)` (해당 요소만 바꾼 새 배열)
  - 삭제: `prev.filter(...)` (조건에 맞춘 새 배열)

~~~typescript
// [app/components/TodoList.tsx]
'use client';

import { useState } from 'react';
import TodoInput from './TodoInput';
import TodoItem, { Todo } from './TodoItem';

export default function TodoList(): React.JSX.Element {
  const [todos, setTodos] = useState<Todo[]>([
    { id: 1, text: 'Next.js 프로젝트 환경 구축하기', completed: true },
    { id: 2, text: '리액트 불변성 원리 학습하기', completed: false },
  ]);

  // 추가: 불변성을 지키며 항목을 추가합니다.
  const handleAddTodo = (text: string) => {
    const newTodo: Todo = { id: Date.now(), text, completed: false };

    // 방법 A: 전개 연산자(...) 활용 (가장 널리 쓰이는 패턴)
    setTodos([...todos, newTodo]);

    // 방법 B: concat() 활용 (동일하게 불변성을 유지하는 새 배열 반환)
    // setTodos(todos.concat(newTodo));
  };

  // 수정: map()을 이용해 지정한 항목만 속성을 변경한 새 배열을 만듭니다.
  const handleToggleTodo = (id: number) => {
    setTodos(
      todos.map((todo) =>
        todo.id === id ? { ...todo, completed: !todo.completed } : todo
      )
    );
  };

  // 삭제: filter()를 이용해 지정한 ID를 제외한 새 배열을 생성합니다.
  const handleDeleteTodo = (id: number) => {
    setTodos(todos.filter((todo) => todo.id !== id));
  };

  return (
    <div className="max-w-md mx-auto my-6 p-6 bg-white border border-gray-200 rounded-xl shadow-sm">
      <h1 className="text-2xl font-bold text-center text-gray-800 mb-6">📝 Todo 리스트</h1>
      <TodoInput onAdd={handleAddTodo} />
      {todos.length === 0 ? (
        <p className="text-center text-gray-500 py-4">등록된 할 일이 없습니다.</p>
      ) : (
        <ul>
          {todos.map((todo) => (
            <TodoItem
              key={todo.id}
              todo={todo}
              onToggle={handleToggleTodo}
              onDelete={handleDeleteTodo}
            />
          ))}
        </ul>
      )}
    </div>
  );
}
~~~

// app/page.tsx
'use client';

import { ChangeEvent, FormEvent, useState, useCallback } from 'react';
import TodoItem from '@/domains/todo/components/TodoItem';

export interface Todo {
    id?: number;
    title?: string;
    checked?: boolean;
}

const initialValue: Todo = {
    title: '',
    checked: false,
};

export default function TodoPage(): React.JSX.Element {
    const [form, setForm] = useState<Todo>(initialValue);
    const [items, setItems] = useState<Todo[]>([]);

    // const onChange = useCallback((e: ChangeEvent<HTMLInputElement>) => {
    //     setForm({ ...form, title: e.target.value });
    // }, [form]); // 두번쨰 매개변수 []는 함수가 새롭게 정의되는 기준 값. [] -> 한번 정의되면 다시 생성 X, [form] -> form 객체가 변화면 useCallback에 정의된 함수가 새로 정의, 생성

    const onChange = useCallback((e: ChangeEvent<HTMLInputElement>) => {
        setForm((prev) => ({ ...prev, title: e.target.value }));
    }, []);

    const onSubmit = useCallback(
        (e: FormEvent<HTMLFormElement>) => {
            e.preventDefault();

            // setItems(items.concat([{ ...form, id: Date.now() }]));
            setItems((prev) => prev.concat([{ ...form, id: Date.now() }]));

            // 등록 완료 이후 값 초기화
            setForm({ ...initialValue });
        },
        [form],
    );

    return (
        <>
            <form onSubmit={onSubmit}>
                <input
                    type="text"
                    name="title"
                    className="border"
                    onChange={onChange}
                    value={form.title}
                />
                <button type="submit" className="border">
                    할일 등록
                </button>
            </form>
            <ul>
                {items.length === 0 ? (
                    <li>등록된 할일이 없습니다.</li>
                ) : (
                    items.map((item) => <TodoItem key={item.id} {...item} />)
                )}
            </ul>
        </>
    );
}

> 교재 · 5장 2강

**Dynamic Routes** `[id]`: 같은 레이아웃을 공유하면서 URL 경로에 따라 다른 데이터를 보여주는 방식

- 폴더명을 대괄호로: `app/products/[id]/page.tsx`
- /products/100 접속 → id 변수에 "100"이 자동 할당
- App Router의 params는 Promise 객체 → `await`로 읽음

~~~typescript
// [app/products/[id]/page.tsx]
// Page Props의 params 타입을 정의합니다.
interface ProductDetailPageProps {
  params: Promise<{
    id: string; // URL 경로에서 전달받은 가변 id 값입니다.
  }>;
}

// 비동기 Server Component로 정의합니다.
export default async function ProductDetailPage({ params }: ProductDetailPageProps) {
  // Promise 형태의 params를 await로 언래핑하여 id를 추출합니다.
  const { id } = await params;

  return (
    <div className="p-4 space-y-2">
      <h1 className="text-xl font-bold">상품 상세 페이지</h1>
      {/* URL 경로에서 추출된 id 파라미터를 그대로 출력합니다. */}
      <p>수신된 상품 ID: <strong className="text-blue-600">{id}</strong></p>
    </div>
  );
}
~~~

**Catch-all Routes** `[...slug]`: 대괄호와 줄임표로 하위 경로를 깊이에 상관없이 모두 감싸 받음

- /shop/clothing/tops 접속 → slug에 `['clothing', 'tops']` 배열 할당
- 뎁스(Depth)가 정해지지 않은 다단계 카테고리를 파일 하나로 처리

~~~typescript
// [app/shop/[...slug]/page.tsx]
// Catch-all params의 타입을 정의합니다.
interface CatchAllPageProps {
  params: Promise<{
    slug: string[]; // 다중 하위 경로가 문자열 배열로 전달됩니다.
  }>;
}

export default async function ShopCategoryPage({ params }: CatchAllPageProps) {
  // Promise 형태의 params를 언래핑합니다.
  const { slug } = await params;

  return (
    <div className="p-4 space-y-2">
      <h1 className="text-xl font-bold">카테고리 경로 분석</h1>
      {/* 수신된 경로의 개수(깊이)를 확인합니다. */}
      <p>경로 깊이: {slug.length}단계</p>
      {/* 배열로 전달받은 slug 항목을 슬래시(/) 구분자로 합쳐서 출력합니다. */}
      <p>전체 경로: <code className="bg-gray-100 p-1">/shop/{slug.join('/')}</code></p>
    </div>
  );
}
~~~

**쿼리 스트링**: URL 끝 물음표(`?key=value`) 뒤의 검색 조건 데이터 — 경로 변경 없이 검색어·정렬·페이지 번호 등을 전달

- `searchParams`는 페이지 컴포넌트 Props로 자동 전달, 값이 없을 수 있어 선택적 속성(?)으로 정의
- params와 똑같이 Promise → await로 해제

~~~typescript
// [app/search/page.tsx]
// searchParams Props의 타입을 정의합니다.
interface SearchPageProps {
  searchParams: Promise<{
    q?: string;      // 검색어 키워드
    sort?: string;   // 정렬 조건
    page?: string;   // 페이지 번호
  }>;
}

export default async function SearchPage({ searchParams }: SearchPageProps) {
  // Promise 형태의 searchParams를 await로 해제합니다.
  const { q, sort, page } = await searchParams;

  return (
    <div className="p-4 space-y-2">
      <h1 className="text-xl font-bold">검색 및 필터링 결과</h1>
      {/* searchParams 객체에서 각 쿼리 스트링 키에 대응하는 값을 가져와 출력합니다. */}
      <ul className="list-disc pl-5 space-y-1">
        <li>검색어(q): <strong>{q || '없음'}</strong></li>
        <li>정렬(sort): <strong>{sort || '기본값'}</strong></li>
        <li>페이지(page): <strong>{page || '1'}</strong></li>
      </ul>
    </div>
  );
}
~~~

// app/todo/page.tsx
'use client';

import { ChangeEvent, FormEvent, useState, useCallback } from 'react';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';

import TodoItem from '@/domains/todo/components/TodoItem';

export interface Todo {
    id?: number;
    title?: string;
    checked?: boolean;
}

const initialValue: Todo = {
    title: '',
    checked: false,
};

export default function TodoPage(): React.JSX.Element {
    const [form, setForm] = useState<Todo>(initialValue);
    const [items, setItems] = useState<Todo[]>([]);

    // const onChange = useCallback((e: ChangeEvent<HTMLInputElement>) => {
    //     setForm({ ...form, title: e.target.value });
    // }, [form]); // 두번쨰 매개변수 []는 함수가 새롭게 정의되는 기준 값. [] -> 한번 정의되면 다시 생성 X, [form] -> form 객체가 변화면 useCallback에 정의된 함수가 새로 정의, 생성

    const onChange = useCallback((e: ChangeEvent<HTMLInputElement>) => {
        setForm((prev) => ({ ...prev, title: e.target.value }));
    }, []);

    const onSubmit = useCallback(
        (e: FormEvent<HTMLFormElement>) => {
            e.preventDefault();

            // setItems(items.concat([{ ...form, id: Date.now() }]));
            setItems((prev) => prev.concat([{ ...form, id: Date.now() }]));

            // 등록 완료 이후 값 초기화
            setForm({ ...initialValue });
        },
        [form],
    );

    const onClick = useCallback((id: number) => {
        setItems((prev) =>
            prev.map((item) =>
                item.id === id ? { ...item, checked: !item.checked } : item,
            ),
        );
    }, []);

    return (
        <div className="m-4">
            <form
                onSubmit={onSubmit}
                className="flex mx-auto w-max-[450px] w-full"
            >
                <Input
                    type="text"
                    name="title"
                    className="border"
                    onChange={onChange}
                    value={form.title}
                />
                <Button type="submit">할일 등록</Button>
            </form>
            <ul>
                {items.length === 0 ? (
                    <li>등록된 할일이 없습니다.</li>
                ) : (
                    items.map((item) => (
                        <TodoItem key={item.id} {...item} onClick={onClick} />
                    ))
                )}
            </ul>
        </div>
    );
}
