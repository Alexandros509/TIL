API ( Application Programming Interface)
  -  애플리케이션 개발 명세서 - 프로그램 간 호출 규약 (Interface ≠ UI)
1.  REST API (Representational State Transfer)
	-  표현적 상태 전이
	-  자원의 **상태**를 JSON 같은 **표현**으로 주고받음
	-  공식: 자원(URI) + 행위(HTTP 메서드)
		-  자원과 행위 
			-  자원: 명사. `/users/3` 
			-  행위: 동사. GET 조회, POST 생성, PUT 전체수정, PATCH 부분수정, DELETE 삭제 
			-  URI (Uniform Resource Identifier): 고유 자원 식별자 
			-  URL: URI 중 위치(주소)를 가리키는 것. `https://api.example.com/users/3`
	-  https://developers.kakao.com/docs/ko/kakaologin/rest-api

저장 프로그램 (Stored Program)
1. 함수 (Function)
	-  매개변수의 반환값
	-  SELECT 구문 내에서 사용 가능 ex) AVG() , SUM()
	-  단일 트랜잭션 : 원자성 O , TCL 사용 불가
2. 프로시저 (Procedure)
	-  독립 프로그램
	-  CALL 프로시저명(...);
	-  트랜잭션 제어 가능 ( COMMIT, ROLLBACK 정의 가능 )
3. 트리거 (Trigger)
	-  추후 다룰 예정

연산자 `:=` : 
-  할당 표현식(Assignment Expression)
-  바다코끼리 연산자(Walrus Operator)
-  반복/조건문 안에서 어떤 값을 계산해 변수에 넣는 동시에, 그 값을 조건식에서 바로 사용할 때 사용

| 기호 | 역할 |
| --- | --- |
| `->>` | JSON 키의 알맹이를 text로 추출 |
| `->` | JSON/JSONB 타입 유지하고 꺼냄 |
| `::` | `CAST(값 AS 타입)`과 같음 (단축 형변환) |
| `UUID` / `INT` | 변환 목표 데이터 타입 |
| `product_id` (맨 뒤) | `AS` 생략한 컬럼 별칭 |

- `p_` = Parameter. 함수 호출 때 밖에서 넘긴 인자  
- `v_` = Variable. `DECLARE`에서 만든 함수 안 임시 변수

~~~python
class Product:
    # 내부 식별자
    _sentinel = object()  # 정적인 클래스 변수

    # 클래스 외부에서 직접 생성은 차단
    def __init__(self, token, name, price, description):
        if token is not Product._sentinel:  # 외부에서 직접 생성한 경우
            raise Exception("직접 생성할 수 없습니다.")

        self.__name = name
        self.__price = price
        self.__description = description

    def name(self):
        return self.__name

    def price(self):
        return self.__price

    def description(self):
        return self.__description

    def __repr__(self):
        return f"Product(name='{self.__name}', price='{self.__price}', description='{self.__description}')"

    # Product 객체를 조립하고 생성하는 클래스
    class Builder:
        def name(self, name):
            self.__name = name
            return self

        def price(self, price):
            self.__price = price
            return self

        def description(self, description):
            self.__description = description
            return self

        # 객체를 생성
        def build(self):
            return Product(Product._sentinel, self.__name, self.__price, self.__description)

    name: str
    price: int = 0
~~~

~~~SQL
CREATE OR REPLACE FUNCTION create\_order\_transaction\(
    p\_order\_name TEXT,
    p\_order\_email TEXT,
    p\_order\_phone TEXT,
    p\_receiver\_name TEXT,
    p\_receiver\_phone TEXT,
    p\_zipcode TEXT,
    p\_address TEXT,
    p\_address\_sub TEXT,
    p\_delivery\_memo TEXT,
    p\_user\_id UUID,
    p\_items JSONB -- \[\{"product\_id": "uuid", "quantity": 1\}, ...\]
\)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER

AS $$
DECLARE
  v\_order\_id UUID;
  v\_order\_no TEXT; -- 20260911113021300
  v\_total\_price INT := 0;
  v\_item RECORD;
  v\_product RECORD;
BEGIN
  -- 상품이 있는지 체크
  IF p\_items IS NULL OR jsonb\_array\_length\(p\_items\) = 0 THEN
    RAISE EXCEPTION '주문할 상품이 없습니다.';
  END IF;

  -- 주문번호 생성
  v\_order\_no := TO\_CHAR\(CURRENT\_TIMESTAMP, 'YYYYMMDDHH24MISS'\);

  -- orders 테이블에 기본 정보를 주문서 추가, 상품 총합 0
  INSERT INTO orders \(
    order\_no, order\_name, order\_email, order\_phone,
    receiver\_name, receiver\_phone, zipcode, address, address\_sub, delivery\_memo,
    total\_price, user\_id
  \) VALUES \(
    v\_order\_no, p\_order\_name, p\_order\_email, p\_order\_phone,
    p\_receiver\_name, p\_receiver\_phone, p\_zipcode, p\_address,
    p\_address\_sub, p\_delivery\_memo, 0, p\_user\_id
  \)
  RETURNING id INTO v\_order\_id;

  -- products 테이블에서 상품을 조회한 후 order\_items 테이블에 상품 정보를 추가
  FOR v\_item IN
      SELECT \(elem->>'product\_id'\)::UUID product\_id,
              \(elem->>'quantity'\)::INT quantity
      FROM jsonb\_array\_elements\(p\_items\) elem
  LOOP
    -- 구매 수량 검증
    IF v\_item.quantity <= 0 THEN
      RAISE EXCEPTION '주문 수량은 1개 이상이어야 합니다.';
    END IF;

    -- 실제 상품 조회
    SELECT id, name, price INTO v\_product
    FROM products WHERE id = v\_item.product\_id AND deleted\_at IS NULL;

    -- 상품 조회의 결과가 없는 경우
    IF NOT FOUND THEN
      RAISE EXCEPTION '등록되지 않거나 삭제된 상품 입니다. \(ID: %\)', v\_item.product\_id;
    END IF;

    -- 주문 상품 등록
    INSERT INTO order\_items \(
      order\_id, product\_id, item\_name, price, quantity
    \) VALUES \(
      v\_order\_id, v\_product.id, v\_product.name, v\_product.price, v\_item.quantity
    \);

    -- total\_price 누적
    v\_total\_price := v\_total\_price + \(v\_product.price \* v\_item.quantity\);
  END LOOP;

  -- orders 테이블에 상품 총금액\(total\_price\)을 업데이트
  UPDATE orders
  SET total\_price = v\_total\_price
  WHERE id = v\_order\_id;

  -- 결과 반환
  RETURN jsonb\_build\_object\(
    'success', TRUE,
    'order\_id', v\_order\_id,
    'order\_no', v\_order\_no
  \);

EXCEPTION
  -- 트랜잭션 실패
  WHEN OTHERS THEN
    RAISE EXCEPTION '주문처리 실패: %', SQLERRM;
END;
$$;
~~~
