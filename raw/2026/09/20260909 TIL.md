# 한 줄로 표현가능 , 가독성 좋게

3.1 로드/불러오기
from langchain_ollama import ChatOllama
# 지시할 수 있는 프롬프트
from langchain_core.prompts import ChatPromptTemplate

3.2  ChatPromptTemplate으로 입력 템플릿 조립하기

# system : 역할, 상황, 제한 조건, 예시 -> 시스템 메세지
# user : 사용자의 질의 : HumanaMessage(..)
# assistant : AI의 답변 : AIMessage(..)

prompt_template = ChatPromptTemplate.from_messages([
    ("system", "당신은 컴퓨터 기초 개념을 일상적인 사물에 빗대어 설명하는 교육 전문가입니다. 2문장 이내로 친절하게 설명하세요."),
    ("user","{user_question}")
])
user_question: "프로그래밍에서 '변수'가 무엇인가요?"
sample_prompt = prompt_template.invoke({
    "user_question" : "프로그래밍에서 '변수'가 무엇인가요?"
})

sample_prompt
3.3 ChatOllama로 모델 호출
model = ChatOllama(
    model="exaone3.5:7.8b-instruct-q4_K_M",
    # base_url="http://localhost::11434" # Ollana 서버 주소
)

model
response = model.invoke(sample_prompt)

response

## 실시간 통신 방식 비교

|**구분**|**Polling (폴링)**|**SSE (Server-Sent Events)**|**WebSocket (웹소켓)**|
|---|---|---|---|
|**통신 방향**|단방향 (`Client → Server`)|**단방향 (`Server → Client`)**|**양방향 (`Client ⇄ Server`)**|

|**자동 재연결**|직접 구현 필요|브라우저 기본 내장 지원|직접 구현 필요|

## Event-Driven (이벤트 기반) 개념

- **개념 정의**  
    - **Polling, SSE, WebSocket**: 데이터를 주고받는 구체적인 **통신 프로토콜/전송 수단**
    - **Event-Driven**: 특정 상태 변화나 사건(Event)이 일어났을 때 이를 감지하고 비동기적으로 반응하여 처리하는 **소프트웨어 설계 패러다임**
- **동작 방식의 차이**
    - **Polling (요청 기반)**: 클라이언트가 변화가 있는지 계속 능동적으로 찔러보는 구조 (`"새 데이터 있어?"`)  
    - **SSE / WebSocket (이벤트 기반 구현체)**: 서버가 대기하고 있다가 내부에서 이벤트가 발행(Publish)되는 순간 연결된 클라이언트로 즉시 밀어넣어 주는(Push) 구조 (`"이벤트 발생했으니 받아가!"`)
